function normalizeSheetKey_(s) { return String(s || '') .replace(/đ/g, 'd').replace(/Đ/g, 'D') .normalize('NFD').replace(/[̀-ͯ]/g, '') .toLowerCase() .replace(/[^a-z0-9]/g, ''); } /** Tìm đúng sheet theo key (VD 'DOC_P1'), thử tên chuẩn trước, nếu không * thấy thì dò theo tên đã chuẩn hoá qua toàn bộ tab đang có trong Sheet. */ function findSheet_(key) { const ss = SpreadsheetApp.getActive(); const configuredName = SHEETS_[key]; let sh = ss.getSheetByName(configuredName); if (sh) return sh; const candidates = (SHEET_NAME_CANDIDATES_[key] || []).concat([normalizeSheetKey_(configuredName)]); sh.getName() : null }; if (sh) { entry.lastRow = sh.getLastRow(); entry.lastCol = sh.getLastColumn(); const rows = sheetToRows_(key); entry.parsedRowCount = rows.length; entry.sampleRow = rows[0] || null; entry.lessonsSeen = Array.from(new Set(rows.map(r => String(r.Lesson)).filter(v => v && v !== 'undefined'))); } info.perKey[key] = entry; }); return info; } /** * API ghi kết quả — giao diện gọi bằng fetch(API_URL, {method:'POST', * body: JSON.stringify({action:'submit', payload:{...}})}). */ function doPost(e) { try { const body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); if (body.action === 'submit') { return jsonOut_(submitResult(body.payload || {})); } return jsonOut_({ ok: false, error: 'Thiếu hoặc sai tham số action (submit).' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Nhiều tên cột khác nhau mà người dùng có thể đã đặt cho cùng 1 ý nghĩa. const LESSON_KEYS_ = ['Lesson', 'Bài', 'Bai', 'bài', 'bai', 'Bài số', 'Bai so', 'Số bài']; const STT_KEYS_ = ['STT', 'Stt', 'stt', 'Số thứ tự', 'So thu tu']; const ANSWER_KEYS_ = ['Dap_an_dung', 'Dap_an', 'Đáp án', 'Dap an', 'Đáp Án', 'DapAn']; const AUDIO_KEYS_ = ['Link_audio', 'Link audio', 'Audio', 'Link_Audio', 'Link nghe', 'Link_nghe']; const IMAGE_KEYS_ = ['Link_anh', 'Link ảnh', 'Anh', 'Link_Anh', 'Link ành', 'Hinh anh', 'Hình ảnh']; const TEXT_KEYS_ = ['Noi_dung', 'Nội dung', 'Noi dung', 'Nội dung sẵn', 'Noi_dung_san', 'Noi dung san']; const PINYIN_KEYS_ = ['Pinyin', 'Phiên âm', 'Phien am']; // Cột "Giải thích" (MỚI): điền lý do vì sao đáp án đúng là đáp án đó — chỉ // hiện cho học sinh khi học sinh làm SAI (hoặc bỏ trống) câu đó, ẩn khi làm // đúng. Đoạn dưới đây tự dò thêm qua * getRichTextValues(): nếu 1 ô có gắn link thật, luôn ưu tiên lấy ĐÚNG LINK * đó, bất kể chữ hiển thị là gì. */ function sheetToRows_(key) { const sh = findSheet_(key); if (!sh || sh.getLastRow() < 2) return []; const range = sh.getDataRange(); const values = range.getValues(); let richTexts = null; try { richTexts = range.getRichTextValues(); } catch (e) { richTexts = null; } const headers = values[0].map(h => String(h).trim()); const normHeaders = headers.map(h => normalizeSheetKey_(h)); const rows = []; for (let i = 1; i < values.length; i++) { const row = values[i]; if (row.every(c => c === '' || c === null)) continue; const obj = {}; const norm = {}; headers.forEach((h, idx) => { let v = row[idx]; if (richTexts) { try { const rt = richTexts[i][idx]; const linkUrl = rt && rt.getLinkUrl && rt.getLinkUrl(); if (linkUrl) v = linkUrl; } catch (e) { /* ô có nhiều đoạn định dạng khác nhau — bỏ qua */ } } obj[h] = v; if (h) norm[normHeaders[idx]] = v; }); obj.__norm = norm; if (obj.Lesson === undefined || obj.Lesson === '') obj.Lesson = pickVal_(obj, LESSON_KEYS_); if (obj.STT === undefined || obj.STT === '') obj.STT = pickVal_(obj, STT_KEYS_); rows.push(obj); } return rows; } function rowsForLesson_(key, baiSo) { return sheetToRows_(key) .filter(r => String(r.Lesson).trim() === String(baiSo).trim()) .sort((a, b) => Number(a.STT) - Number(b.STT)); } function firstNonEmpty_(rows, key) { for (const r of rows) { const v = pickVal_(r, [key]); if (v !== undefined && v !== '') return v; } return ''; } function firstNonEmptyAny_(rows, keys) { for (const key of keys) { const v = firstNonEmpty_(rows, key); if (v) return v; } return ''; } /** * Cô giáo có thể GỘP Ô (merge cell) cột "Giải thích" cho cả 1 khối nhiều câu * (ví dụ gộp STT 6-8 làm 1 ô, rồi gõ hết giải thích cho 3 câu vào đó, mỗi câu * bắt đầu bằng số thứ tự riêng, ví dụ: * "6.
